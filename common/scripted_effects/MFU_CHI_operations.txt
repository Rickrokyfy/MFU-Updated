MFU_CHI_daily = {

    if = {
        limit = { MFU_CHI_full_scope = yes }
        if = { limit = { NOT = { has_country_flag = MFU_CHI_initialized } } MFU_CHI_initialize = yes }
        if = { limit = { check_variable = { MFU_CHI_recovery_days > 0 } } subtract_from_variable = { MFU_CHI_recovery_days = 1 } }
        if = { limit = { check_variable = { MFU_CHI_window_days > 0 } } subtract_from_variable = { MFU_CHI_window_days = 1 } }
        if = { limit = { check_variable = { MFU_CHI_request_days > 0 } } subtract_from_variable = { MFU_CHI_request_days = 1 } }
        if = {
            limit = {
                check_variable = { MFU_CHI_request > 0 }
                OR = {
                    NOT = { check_variable = { MFU_CHI_request_days > 0 } }
                    NOT = { MFU_CHI_USA_interface_enabled = yes }
                    USA = { OR = { exists = no is_ai = yes has_war_with = CHI } }
                    AND = { check_variable = { MFU_CHI_request = 1 } NOT = { MFU_CHI_target_609_valid = yes } }
                    AND = { check_variable = { MFU_CHI_request = 2 } NOT = { MFU_CHI_target_597_valid = yes } }
                }
            }
            MFU_CHI_clear_request = yes
        }
        if = {
            limit = {
                check_variable = { MFU_CHI_phase = 2 }
                OR = {
                    NOT = { check_variable = { MFU_CHI_window_days > 0 } }
                    NOT = { MFU_CHI_objective_valid = yes }
                    check_variable = { MFU_CHI_request = 3 }
                }
            }
            MFU_CHI_recover = yes
        }
    }
    else_if = { limit = { has_country_flag = MFU_CHI_initialized } MFU_CHI_cleanup = yes }
}
MFU_CHI_weekly = {

    if = {
        limit = { MFU_CHI_full_scope = yes has_country_flag = MFU_CHI_initialized }
        MFU_CHI_measure_readiness = yes
        if = {
            limit = { check_variable = { MFU_CHI_phase = 2 } }
            if = {
                limit = { NOT = { has_country_flag = MFU_CHI_hold_ready } }
                MFU_CHI_recover = yes
            }
        }
        else_if = {
            limit = {
                NOT = { check_variable = { MFU_CHI_recovery_days > 0 } }
                NOT = { check_variable = { MFU_CHI_request = 3 } }
                has_country_flag = MFU_CHI_entry_ready
                OR = { MFU_CHI_target_609_valid = yes MFU_CHI_target_597_valid = yes }
            }
            MFU_CHI_select_objective = yes
            set_variable = { MFU_CHI_phase = 1 }
            add_to_variable = { MFU_CHI_prepared_weeks = 1 }
            if = {
                limit = { check_variable = { MFU_CHI_prepared_weeks > 1 } }
                set_variable = { MFU_CHI_phase = 2 }
                set_variable = { MFU_CHI_window_days = 14 }
                set_variable = { MFU_CHI_prepared_weeks = 0 }
            }
        }
        else = {
            set_variable = { MFU_CHI_phase = 0 }
            set_variable = { MFU_CHI_prepared_weeks = 0 }
            set_variable = { MFU_CHI_objective = 0 }
        }
    }
